const form = document.querySelector('form');
form.addEventListener('submit', function (e) {
  e.preventDefault();
  const height=parseInt(document.querySelector('#height').value); 
  //for converting string to int parseint is used
  const weight=parseInt(document.querySelector('#weight').value);
  const result=document.querySelector('#results');

  if(height === '' || height < 0 || isNaN(height) ){
    result.innerHTML=`please enter a valid input for height having value ${height}`;
  } 
  else if(weight === '' || weight < 0 || isNaN(weight) ){
    result.innerHTML=`please enter a valid input for weight having value ${weight}`;
  }
  else{
    result.innerHTML=(weight /((height*height)/10000)).toFixed(2);
    const bmi=parseFloat((weight /((height*height)/10000)).toFixed(2));
    if(bmi < 18.6){
      result.innerHTML=`it is underweight , ${height} and ${weight} are their height and weight`;
    }
    else if(bmi >= 18 || bmi <= 24.9){
      result.innerHTML=`it is in normal range , ${height} and ${weight} are their height and weight`;
    }
    else{
      result.innerHTML=`it is overweight range , ${height} and ${weight} are their height and weight`;
    }
  }
});
